## `controller/authController.py`

### `AuthController.login(payload)`

<div class="pseudocode">
LOGIN(PAYLOAD)

    USER <- CALL AUTH_SERVICE.AUTHENTICATE_USER(PAYLOAD.IDENTIFIER, PAYLOAD.PASSWORD)

    ACCESS_TOKEN, EXPIRES_IN <- CALL AUTH_SERVICE.CREATE_ACCESS_TOKEN(USER.ID, USER.USERNAME, USER.ROLE)

    REFRESH_TOKEN, REFRESH_EXPIRES_IN <- CALL AUTH_SERVICE.CREATE_REFRESH_TOKEN(USER.ID)

    RESPONSE <- TOKEN_RESPONSE(ACCESS_TOKEN, EXPIRES_IN, REFRESH_TOKEN, REFRESH_EXPIRES_IN, USER)

    RETURN RESPONSE
</div>

### `AuthController.forgot_password(payload)`

<div class="pseudocode">
FORGOT_PASSWORD(PAYLOAD)

    IF PAYLOAD.EMAIL IS EMPTY

        RETURN HTTP_422("FIELD EMAIL TIDAK BOLEH KOSONG")

    EMAIL <- TRIM(PAYLOAD.EMAIL)

    MESSAGE <- CALL AUTH_SERVICE.REQUEST_PASSWORD_RESET(EMAIL)

    RESPONSE <- MESSAGE_RESPONSE(MESSAGE)

    RETURN RESPONSE
</div>

<div class="pseudocode">
VERIFY_RESET_PIN(PAYLOAD)

### `AuthController.reset_password(payload)`

<div class="pseudocode">
RESET_PASSWORD(PAYLOAD)

    IF LENGTH(PAYLOAD.NEW_PASSWORD) < 8

        RETURN HTTP_422("PASSWORD BARU MINIMAL 8 KARAKTER")

        RETURN HTTP_422("FIELD RESET_TOKEN TIDAK BOLEH KOSONG")

    MESSAGE <- CALL AUTH_SERVICE.RESET_PASSWORD(RESET_TOKEN, PAYLOAD.NEW_PASSWORD)

    RESPONSE <- MESSAGE_RESPONSE(MESSAGE)

    RETURN RESPONSE

## `controller/chatController.py`

### `ChatController.handle_chat(request, current_user)`

<div class="pseudocode">
HANDLE_CHAT(REQUEST, CURRENT_USER)

    USER_ID <- CURRENT_USER.ID

    ROLE_VALUE <- EXTRACT_ROLE_VALUE(CURRENT_USER)

    USER_ROLE <- TO_CHAT_ROLE(ROLE_VALUE)

    USER_DIVISI <- NULL

    IF USER_ID IS EMPTY OR USER_ROLE IS EMPTY

        RETURN HTTP_401("TOKEN TIDAK MEMUAT INFORMASI USER YANG VALID")

    RESPONSE <- CALL CHAT_SERVICE.PROCESS_QUERY(REQUEST.MESSAGE, USER_ID, USER_ROLE, USER_DIVISI, REQUEST.SESSION_ID, REQUEST.SHOW_SQL)

    STREAMING_RESPONSE <- BUILD_STREAMING_RESPONSE(RESPONSE)

    RETURN STREAMING_RESPONSE
</div>

### `ChatController.handle_clarification(request, current_user)`

<div class="pseudocode">
HANDLE_CLARIFICATION(REQUEST, CURRENT_USER)

    IF REQUEST.SESSION_ID IS EMPTY OR REQUEST.CLARIFICATION_ANSWER IS EMPTY

        RETURN HTTP_400("SESSION_ID DAN CLARIFICATION_ANSWER WAJIB DISERTAKAN")

    USER_ID <- CURRENT_USER.ID

    ROLE_VALUE <- EXTRACT_ROLE_VALUE(CURRENT_USER)

    USER_ROLE <- TO_CHAT_ROLE(ROLE_VALUE)

    USER_DIVISI <- NULL

    IF USER_ID IS EMPTY OR USER_ROLE IS EMPTY

        RETURN HTTP_401("TOKEN TIDAK MEMUAT INFORMASI USER YANG VALID")

    DISAMBIGUATION_RESULT <- CALL CLARIFICATION_SERVICE.HANDLE_CLARIFICATION_RESPONSE(REQUEST.SESSION_ID, REQUEST.CLARIFICATION_ANSWER)

    IF DISAMBIGUATION_RESULT NOT FOUND

        RETURN HTTP_404(ERROR_MESSAGE)

    RESPONSE <- CALL CHAT_SERVICE.PROCESS_QUERY(DISAMBIGUATION_RESULT.DISAMBIGUATED_QUERY, USER_ID, USER_ROLE, USER_DIVISI, REQUEST.SESSION_ID, REQUEST.SHOW_SQL, DISAMBIGUATION_RESULT)

    STREAMING_RESPONSE <- BUILD_STREAMING_RESPONSE(RESPONSE)

    RETURN STREAMING_RESPONSE
</div>

### `KPIGroupController.list_groups(page, limit, tahun, group_type, search)`

<div class="pseudocode">
LIST_GROUPS(PAGE, LIMIT, TAHUN, GROUP_TYPE, SEARCH)

    IF LIMIT < 1 OR LIMIT > 100

        RETURN HTTP_422("LIMIT HARUS ANTARA 1 DAN 100")

        RETURN HTTP_422("PAGE MINIMAL 1")

    ROWS, TOTAL <- CALL KPI_GROUP_SERVICE.LIST_GROUPS(PAGE, LIMIT, TAHUN, GROUP_TYPE, SEARCH)

    TOTAL_PAGES <- CEIL(TOTAL / LIMIT)

    DATA <- MAP_EACH(ROWS, KPI_GROUP_RESPONSE_WITHOUT_RECORDS)

    RESPONSE <- KPI_GROUP_LIST_RESPONSE(TOTAL, PAGE, LIMIT, TOTAL_PAGES, DATA)

    RETURN RESPONSE

### `UserController.get_all_users(page, limit, search, role, user_status)`

<div class="pseudocode">
GET_ALL_USERS(PAGE, LIMIT, SEARCH, ROLE, USER_STATUS)

    IF LIMIT < 1 OR LIMIT > 100

        RETURN HTTP_422("LIMIT HARUS ANTARA 1 DAN 100")

    IF PAGE < 1

        RETURN HTTP_422("PAGE MINIMAL 1")

    ACTIVE_STATUS_FILTER <- NULL

    IF USER_STATUS IS NOT EMPTY

        ACTIVE_STATUS_FILTER <- USER_STATUS == ACTIVE

    RESULT <- CALL USER_SERVICE.GET_ALL_USERS(PAGE, LIMIT, SEARCH, ROLE, ACTIVE_STATUS_FILTER)

    USERS <- MAP_EACH(RESULT.USERS, USER_RESPONSE)

    RESPONSE <- DICTIONARY("TOTAL", RESULT.TOTAL, "PAGE", RESULT.PAGE, "LIMIT", RESULT.LIMIT, "USERS", USERS)

    RETURN RESPONSE
</div>
