### `AuthService.rotate_tokens(refresh_token)`

<div class="pseudocode">
ROTATE_TOKENS(REFRESH_TOKEN)

    PAYLOAD <- CALL AUTH_SERVICE.DECODE_REFRESH_TOKEN(REFRESH_TOKEN)

    IF REFRESH_TOKEN IS REVOKED

        RETURN HTTP_401("REFRESH TOKEN SUDAH DICABUT")

    USER <- CALL USER_REPOSITORY.GET_BY_ID(PAYLOAD.SUB)

    IF USER NOT FOUND OR USER IS INACTIVE

        RETURN HTTP_401("USER TIDAK VALID")

    CALL AUTH_SERVICE.REVOKE_REFRESH_TOKEN(REFRESH_TOKEN)

    ACCESS_TOKEN, ACCESS_EXP <- CALL AUTH_SERVICE.CREATE_ACCESS_TOKEN(USER.ID, USER.USERNAME, USER.ROLE)

    NEW_REFRESH_TOKEN, REFRESH_EXP <- CALL AUTH_SERVICE.CREATE_REFRESH_TOKEN(USER.ID)

    RETURN ACCESS_TOKEN, ACCESS_EXP, NEW_REFRESH_TOKEN, REFRESH_EXP
</div>

### `AuthService.verify_reset_pin(email, pin)`

<div class="pseudocode">
VERIFY_RESET_PIN(EMAIL, PIN)

    USER <- CALL USER_REPOSITORY.GET_BY_EMAIL(EMAIL)

    IF USER NOT FOUND

        RETURN HTTP_400("PIN TIDAK VALID")

    RESET_RECORD <- CALL PASSWORD_RESET_REPOSITORY.GET_ACTIVE_BY_USER_ID(USER.ID)

    IF RESET_RECORD NOT FOUND OR RESET_RECORD EXPIRED

        RETURN HTTP_400("PIN TIDAK VALID ATAU KEDALUWARSA")

    IS_VALID <- CALL AUTH_SERVICE.VERIFY_PASSWORD(PIN, RESET_RECORD.PIN_HASH)

    IF IS_VALID IS FALSE

        RETURN HTTP_400("PIN TIDAK VALID")

    CALL PASSWORD_RESET_REPOSITORY.MARK_USED(RESET_RECORD.ID)

    RESET_TOKEN <- CALL JWT.ENCODE(RESET_TOKEN_PAYLOAD, RESET_SECRET_KEY, ALGORITHM)

    RETURN RESET_TOKEN, EXPIRES_IN
</div>

INGEST_KPI_MASTER(SHEET_URL, TAHUN)

## `service/TrackeringestionService.py`

### `TrackerIngestionService.ingest_all_sheets(sheet_url, tahun, nama_orang_override, skip_on_error)`

<div class="pseudocode">
INGEST_ALL_SHEETS(SHEET_URL, TAHUN, NAMA_ORANG_OVERRIDE, SKIP_ON_ERROR)

    SHEETS <- CALL GOOGLE_SHEET_SERVICE.FETCH_ALL_SHEETS_AS_DATAFRAMES(SHEET_URL, SKIP_ON_ERROR)

    RESULTS <- EMPTY_LIST

    FOR EACH SHEET IN SHEETS

        GROUP <- CALL KPI_GROUP_REPOSITORY.UPSERT_TRACKER_GROUP(SHEET_URL, SHEET.NAME, TAHUN)

        RECORDS <- PARSE_TRACKER_DATAFRAME(SHEET.DATAFRAME, GROUP.ID, TAHUN, NAMA_ORANG_OVERRIDE)

        INGEST_RESULT <- CALL KPI_TRACKER_REPOSITORY.UPSERT_RECORDS(RECORDS)

        APPEND INGEST_RESULT TO RESULTS

    SUMMARY <- BUILD_BULK_INGESTION_SUMMARY(RESULTS)

    RETURN SUMMARY
</div>

### `TrackerIngestionService.ingest_batch(sources, skip_on_error, delay_between_sources)`

<div class="pseudocode">
INGEST_BATCH(SOURCES, SKIP_ON_ERROR, DELAY_BETWEEN_SOURCES)

    RESULTS <- EMPTY_LIST

        RESULT <- CALL TRACKER_INGESTION_SERVICE.INGEST_ALL_SHEETS(SOURCE.SHEET_URL, SOURCE.TAHUN, SOURCE.NAMA_ORANG_OVERRIDE, SKIP_ON_ERROR)

        APPEND RESULT TO RESULTS

    SUMMARY <- BUILD_BATCH_INGESTION_SUMMARY(RESULTS)

FETCH_SHEET_AS_DATAFRAME(SHEET_URL, SHEET_NAME)

### `GoogleSheetService.fetch_all_sheets_as_dataframes(sheet_url, skip_on_error)`

FETCH_ALL_SHEETS_AS_DATAFRAMES(SHEET_URL, SKIP_ON_ERROR)

    SPREADSHEET_ID <- EXTRACT_SPREADSHEET_ID(SHEET_URL)

    WORKSHEETS <- CALL GOOGLE_SHEETS_CLIENT.OPEN_BY_KEY(SPREADSHEET_ID).WORKSHEETS()

    RESULTS <- EMPTY_LIST

    FOR EACH WORKSHEET IN WORKSHEETS

        TRY

            DATAFRAME, METADATA <- CALL GOOGLE_SHEET_SERVICE.FETCH_SHEET_AS_DATAFRAME(SHEET_URL, WORKSHEET.TITLE)

            APPEND SHEET_RESULT TO RESULTS

        CATCH ERROR

            IF SKIP_ON_ERROR IS FALSE

                RAISE ERROR

            APPEND FAILED_SHEET_RESULT TO RESULTS

    RETURN RESULTS

## `service/schedulerJobService.py`

### `SchedulerJobService.run_ingestion_job()`

<div class="pseudocode">
RUN_INGESTION_JOB()

    CONFIG <- CALL SCHEDULER_CONFIG_REPOSITORY.GET_CONFIG()

    IF CONFIG.IS_ACTIVE IS FALSE

        RETURN VOID

    GROUPS <- CALL KPI_GROUP_REPOSITORY.GET_ACTIVE_SCHEDULED_GROUPS()

    FOR EACH GROUP IN GROUPS

        IF GROUP.GROUP_TYPE == "MASTER"

            CALL KPI_MASTER_INGESTION_SERVICE.INGEST_KPI_MASTER(GROUP.SHEET_URL, GROUP.TAHUN)

        ELSE IF GROUP.GROUP_TYPE == "TRACKER"

            CALL TRACKER_INGESTION_SERVICE.INGEST_ALL_SHEETS(GROUP.SHEET_URL, GROUP.TAHUN, NULL, TRUE)

        WAIT CONFIG.DELAY_BETWEEN_GROUPS SECONDS

    CALL SCHEDULER_JOB_SERVICE.AUTO_PAUSE_IF_DECEMBER(CONFIG)

    RETURN VOID
</div>
